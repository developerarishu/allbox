# ================= POSTGRESQL DATABASE LAYER =================
def get_db_connection():
    return psycopg2.connect(DATABASE_URL)

def init_db():
    try:
        with get_db_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute("""
                    CREATE TABLE IF NOT EXISTS users (
                        user_id BIGINT PRIMARY KEY,
                        username TEXT,
                        first_name TEXT,
                        status TEXT DEFAULT 'pending',
                        warnings INTEGER DEFAULT 0,
                        prompts_count INTEGER DEFAULT 0,
                        joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                    )
                """)
                cursor.execute("""
                    ALTER TABLE users 
                    ADD COLUMN IF NOT EXISTS current_mode TEXT DEFAULT 'chat';
                """)
            conn.commit()
    except Exception as e:
        logging.error(f"Database Initialization Error: {e}")

def get_user(user_id):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute("SELECT user_id, username, first_name, status, warnings, prompts_count FROM users WHERE user_id = %s", (user_id,))
            return cursor.fetchone()

def get_target_uid(arg):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            if arg.isdigit():
                cursor.execute("SELECT user_id FROM users WHERE user_id = %s", (int(arg),))
            else:
                uname = arg.replace("@", "")
                cursor.execute("SELECT user_id FROM users WHERE username ILIKE %s", (uname,))
            res = cursor.fetchone()
            return res[0] if res else None

def upsert_user(user_id, username, first_name):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            user = get_user(user_id)
            if not user:
                initial_status = 'approved' if user_id == ADMIN_ID else 'pending'
                cursor.execute("""
                    INSERT INTO users (user_id, username, first_name, status) 
                    VALUES (%s, %s, %s, %s)
                """, (user_id, username, first_name, initial_status))
                conn.commit()
                return initial_status
            else:
                cursor.execute("UPDATE users SET username = %s, first_name = %s WHERE user_id = %s", (username, first_name, user_id))
                conn.commit()
                return user[3]

def set_user_status(user_id, status):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute("UPDATE users SET status = %s WHERE user_id = %s", (status, user_id))
        conn.commit()

def increment_prompts(user_id):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute("UPDATE users SET prompts_count = prompts_count + 1 WHERE user_id = %s RETURNING prompts_count", (user_id,))
            updated_count = cursor.fetchone()[0]
        conn.commit()
        return updated_count

def add_warning(user_id):
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute("UPDATE users SET warnings = warnings + 1 WHERE user_id = %s", (user_id,))
            cursor.execute("SELECT warnings FROM users WHERE user_id = %s", (user_id,))
            res = cursor.fetchone()
        conn.commit()
        return res[0] if res else 0

def get_all_users():
    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            cursor.execute("SELECT user_id, username, first_name, status, prompts_count, warnings FROM users")
            return cursor.fetchall()

def set_user_mode(user_id, mode):

async def prepare_image_payload(photo_array, caption_text):
    file_obj = await photo_array[-1].get_file()
    img_bytearray = await file_obj.download_as_bytearray()
    b64_data = base64.b64encode(img_bytearray).decode('utf-8')
    text_prompt = caption_text if caption_text else "Please thoroughly inspect and read this screenshot/image. If it's an error, debug it. If it has code, transcribe/fix it. Provide full details."
    return [{"type": "image", "source": {"type": "base64", "media_type": "image/jpeg", "data": b64_data}}, {"type": "text", "text": text_prompt}]

    if current_status == "trial" and user_prompts >= TRIAL_PROMPTS_LIMIT:
        set_user_status(user.id, "pending")
        await update.message.reply_text(
            f"⌛ *Free Trial Expired!*\n\nTumhare *{TRIAL_PROMPTS_LIMIT} free prompts* khatam ho chuke hain.\nFull access lene ke liye support par sampark karein: @CLAUDASUPPORTBOT",
            parse_mode=ParseMode.MARKDOWN
        )
        return

    is_photo = bool(update.message.photo)
    user_text = update.message.caption if is_photo else update.message.text
    if not is_photo and not user_text: return

    new_count = increment_prompts(user.id)

    trial_notice = ""
    if current_status == "trial":
        remaining = max(0, TRIAL_PROMPTS_LIMIT - new_count)
        trial_notice = f"\n`[Trial: {remaining} prompts left]`"

    # Fetch User Mode
    user_mode = get_user_mode(user.id)
    
    if user_mode == "chat":
        system_instruction = CHAT_PROMPT
        timeout_val = 600.0  # 10 minutes limit for chat
        loading_text = "💭 *Opus 4.6 is thinking...*"
    else:
        system_instruction = CODE_PROMPT
        timeout_val = None   # No limit for max project compile wait
        loading_text = "⚙️ *Code Engine is building...* \n_(Zero fluff, pure code. Large projects may take time.)_"

    loading_msg = await update.message.reply_text(loading_text, parse_mode=ParseMode.MARKDOWN)
    await context.bot.send_chat_action(chat_id=chat_id, action='typing')
    start_time = time.time()

    try:
        if is_photo:
            user_content = await prepare_image_payload(update.message.photo, user_text)
        else:
            user_content = user_text

        history = get_memory(chat_id)
        history.append({"role": "user", "content": user_content})
        if len(history) > MAX_HISTORY: history = history[-MAX_HISTORY:]

        # ===== BATTLE-TESTED API CALL (Connection Crash Fixed) =====
        try:
            api_task = ai_client.messages.create(
                model=MODEL_NAME,
                max_tokens=4000,
                system=system_instruction,
                messages=history
                # No timeout parameter passed here to prevent httpx dropping the connection
            )
            response = await asyncio.wait_for(api_task, timeout=timeout_val)
            
        except asyncio.TimeoutError:
            err = "❌ *Timeout Error:* Noah API ka server hang ho gaya hai. Kripya thodi der baad prompt bhejein." + SHOUTOUT
            await loading_msg.edit_text(err, parse_mode=ParseMode.MARKDOWN)
            if history and history[-1]["role"] == "user": history.pop()
            return 
        except anthropic.APIStatusError as e:
            error_str = str(e)
            if "529" in error_str or "overloaded_error" in error_str:
                err = "⚠️ *Server Overloaded (529):* Noah API backend currently busy hai. Kripya 2-3 minute baad try karein." + SHOUTOUT
            else:
                err = f"❌ *API Error:* `{e.status_code}` - Request failed." + SHOUTOUT
            await loading_msg.edit_text(err, parse_mode=ParseMode.MARKDOWN)
            if history and history[-1]["role"] == "user": history.pop()
            return
        except Exception as e:
            err = f"❌ *Connection/System Error:* `{str(e)}`" + SHOUTOUT
            await loading_msg.edit_text(err, parse_mode=ParseMode.MARKDOWN)
            if history and history[-1]["role"] == "user": history.pop()
            return
        # ================================

        ai_reply = response.content[0].text
        history.append({"role": "assistant", "content": ai_reply})
        elapsed = round(time.time() - start_time, 1)

        # Tokens Details
        try:
            in_tokens = response.usage.input_tokens
            out_tokens = response.usage.output_tokens
            total_tokens = in_tokens + out_tokens
        except:
            in_tokens = out_tokens = total_tokens = 0

        # ===== SUPER SMART FILE EXTRACTION =====
       # ===== SUPER SMART FILE EXTRACTION (Fixed Incomplete Blocks) =====
        if user_mode == "code":
            # Naya Regex: (?:```|\Z) ka matlab hai ya toh closing ``` dhoondo, ya End-of-Message tak utha lo
            code_blocks = re.findall(r"(```([a-zA-Z0-9+#_\s]*)\r?\n(.*?)(?:```|\Z))", ai_reply, re.DOTALL)
            
            if code_blocks:
                counter = 1
                for full_match, lang_info, code in code_blocks:
                    # Bhasha nikaalne ka try karo (agar empty hua toh koi baat nahi)
                    lang = lang_info.strip().split()[0].lower() if lang_info.strip() else ""
                    
                    # Pehli line se actual File Name dhundho
                    first_line = code.strip().split('\n')[0]
                    name_match = re.search(r'([a-zA-Z0-9_\-\.]+\.[a-zA-Z0-9]+)', first_line)
                    
                    file_name = ""
                    ext = ""
                    if name_match:
                        file_name = name_match.group(1)
                        ext = file_name.split('.')[-1].lower()
                    else:
                        ext = "py" if lang == "python" else ("js" if lang == "javascript" else ("html" if lang == "html" else lang))
                        if not ext: ext = "txt"
                        file_name = f"script_{counter}.{ext}"
                    
                    # SKIP LOGIC (Terminal commands aur random text ko file nahi banayega)
                    if ext in ["bash", "sh", "shell"]: 
                        continue
                    if len(code.strip()) < 300 and not name_match: 
                        continue
                    if ext in ["txt", "text", "plaintext"] and not name_match: 
                        continue
                    
                    # Legitimate code block hai! File banao!
                    try:
                        vfile = io.BytesIO(code.encode('utf-8'))
                        vfile.name = file_name
                        await context.bot.send_document(chat_id=chat_id, document=vfile, caption=f"📁 `{file_name}` Ready!")
                        counter += 1
                        
                        # PERFECT REPLACE: Chat se raw text ekdum clean!
                        ai_reply = ai_reply.replace(full_match, f"\n*[ 📁 `{file_name}` upar attach kar di gayi hai ]*\n")
                    except Exception as file_err:
                        logging.error(f"File upload failed: {file_err}")

            meta_info = f"\n\n`[⏱️ {elapsed}s | 🪙 Tokens: In {in_tokens} + Out {out_tokens} = {total_tokens}]`"
        else:
            meta_info = f"\n\n`[⏱️ {elapsed}s]`"
        # =================================================

        final_msg = ai_reply + meta_info + trial_notice + SHOUTOUT
        
        # Safe chunking & sending
        if len(final_msg) > 4000:
            chunks = [final_msg[i:i+3900] for i in range(0, len(final_msg), 3900)]
            await loading_msg.edit_text(chunks[0] + "...")
            for chunk in chunks[1:]:
                await context.bot.send_message(chat_id=chat_id, text=chunk)
        else:
            try:
                await loading_msg.edit_text(final_msg, parse_mode=ParseMode.MARKDOWN)
            except Exception:
                await loading_msg.edit_text(final_msg)

    except Exception as e:
        err = f"❌ *System Error:* `{str(e)}`" + SHOUTOUT
        try:
            await loading_msg.edit_text(err, parse_mode=ParseMode.MARKDOWN)
        except Exception:
            pass 
        if history and history[-1]["role"] == "user": history.pop()
